cmake_minimum_required(VERSION 3.19)
if(NOT with-nest)
  message(FATAL_ERROR "Pass -Dwith-nest=/path/to/nest-config")
endif()
execute_process(COMMAND ${with-nest} --compiler OUTPUT_VARIABLE NEST_COMPILER OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(DEFINED ENV{NEST_CXX})
 set(CMAKE_CXX_COMPILER "$ENV{NEST_CXX}")
elseif(EXISTS "${NEST_COMPILER}")
 set(CMAKE_CXX_COMPILER "${NEST_COMPILER}")
else()
 message(WARNING "nest-config compiler '${NEST_COMPILER}' is missing. CMake will use the system C++ compiler; override with NEST_CXX if needed.")
endif()
project(pwa_astrocyte_module LANGUAGES CXX)
foreach(flag IN ITEMS cflags includes libs)
 execute_process(COMMAND ${with-nest} --${flag} OUTPUT_VARIABLE NEST_${flag} OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
 separate_arguments(NEST_${flag} NATIVE_COMMAND "${NEST_${flag}}")
endforeach()
# Pip wheels may retain build-machine paths in nest-config. Filter missing
# absolute paths and use actual include locations discovered by build_module.sh.
set(FIXED_NEST_LIBS "")
foreach(item IN LISTS NEST_libs)
 if(item MATCHES "^-L(.+)$")
  if(IS_DIRECTORY "${CMAKE_MATCH_1}")
   list(APPEND FIXED_NEST_LIBS "${item}")
  endif()
 elseif(IS_ABSOLUTE "${item}" AND NOT EXISTS "${item}")
  if(item MATCHES "/libpthread\\.so$")
   list(APPEND FIXED_NEST_LIBS -pthread)
  endif()
 else()
  list(APPEND FIXED_NEST_LIBS "${item}")
 endif()
endforeach()
add_library(pwa_astrocyte_module MODULE module.cpp bridge.cpp pwa_event.cpp generated/astrocyte_am_pwa.cpp generated/astrocyte_fm_pwa.cpp)
option(PWA_PROFILE_ENABLED "Instrument PWA solver calls (changes performance)" OFF)
if(PWA_PROFILE_ENABLED)
 target_compile_definitions(pwa_astrocyte_module PRIVATE PWA_PROFILE=1)
endif()
target_compile_options(pwa_astrocyte_module PRIVATE ${NEST_cflags} ${NEST_includes} -DLTX_MODULE)
if(DEFINED NEST_GLIBCXX_ABI AND NOT NEST_GLIBCXX_ABI STREQUAL "")
 target_compile_definitions(pwa_astrocyte_module PRIVATE _GLIBCXX_USE_CXX11_ABI=${NEST_GLIBCXX_ABI})
endif()
target_link_options(pwa_astrocyte_module PRIVATE ${FIXED_NEST_LIBS})
target_include_directories(pwa_astrocyte_module PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(NEST_INCLUDE_DIR)
 target_include_directories(pwa_astrocyte_module PRIVATE "${NEST_INCLUDE_DIR}")
endif()
if(BOOST_INCLUDE_DIR)
 target_include_directories(pwa_astrocyte_module PRIVATE "${BOOST_INCLUDE_DIR}")
endif()
if(EXTRA_INCLUDE_DIR AND IS_DIRECTORY "${EXTRA_INCLUDE_DIR}")
 target_include_directories(pwa_astrocyte_module PRIVATE "${EXTRA_INCLUDE_DIR}")
endif()
set_target_properties(pwa_astrocyte_module PROPERTIES PREFIX "")
